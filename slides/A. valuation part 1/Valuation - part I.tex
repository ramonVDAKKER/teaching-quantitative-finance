\section{Introduction}

\begin{frame}{Options (prerequisites)}
We assume familiarity with the basics of options - calls and puts, payoffs at maturity, payoff diagrams, moneyness, and exercise styles. These are collected in the companion deck \emph{``Options: a quick recap''}.
\medskip

Throughout, $S_t$ denotes the price of the underlying at time $t$, $K$ the strike, and $T$ the maturity; a European payoff is written $C_T=g(S_T)$.
\medskip

\textbf{Notation.} We write $x^+:=\max\{x,0\}$ for the positive part of $x$. So the payoff of a call is $(S_T-K)^+$ and that of a put is $(K-S_T)^+$.
\end{frame}

\begin{frame}{Main problems}
An option has no obvious ``correct'' price: its value is \emph{derived} from the underlying. This raises two central problems.
\medskip

\textbf{1.\ Valuation/pricing.} What is the `fair' price $C_t$ of an option, and how does it evolve over time? This matters for:
\begin{itemize}
\item investors/traders - to avoid overpaying, or to profit from a market price they believe is wrong;
\item financial institutions - to quote prices for OTC (or `embedded') options, and to value and risk-manage books of options.
\end{itemize}
\medskip

\textbf{2.\ Hedging.} Having sold an option, how do you control the resulting risk? (Treated later in the course.)
\medskip

A striking feature of the models we use: these two problems turn out to share the \emph{same} solution.
\end{frame}

\begin{frame}{Main problems - valuation}
\begin{itemize}
\item we are interested in (fair) price $C_t$, for $t\in [0,T)$, of (European) option with payoff $C_T=g(S_T)$ at maturity $T$
\item we will learn three methods to determine `fair price' (using specified continuous-time model for financial market):
\begin{itemize}
\item Black-Scholes Partial Differential Equation
\item risk-neutral pricing
\item pricing kernel
\end{itemize}
\item in particular: you will derive the famous Black-Scholes price for European put option with maturity $T$ and strike $K$:
{%\small
\begin{align*}
C_t&=p(S_t,T-t,K,r,\sigma)= K\e^{-r (T-t)} \Phi(-d_2) -    S_t  \Phi(-d_1)
\\
d_1&=\frac{ \log\left(\frac{S_t}{K}\right) + \left( r+ \frac{\sigma^2}{2}\right) (T-t)}{ \sigma \sqrt{T-t}}, \quad d_2=d_1-\sigma \sqrt{T-t},
\end{align*}
}
\begin{itemize}
\item does not depend on $\mu$!
\end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Arbitrage}
All pricing methods we discuss rely on insisting that arbitrage opportunities do/should not
exist.

\begin{definition}
If it is possible to find an admissible self-financing trading strategy with value process $V$ such that, for some  $T>0$, we have
$V_0\leq 0$ and
\[
\mathbb{P}( V_T< 0 )=0 \text{ and }  \mathbb{P}( V_T> 0 )>0,
\]
then we say that an arbitrage opportunity exists.
\end{definition}
\textbf{Remarks:}
\begin{itemize}
\item in QF you are allowed to ignore `admissible': this condition rules out `doubling strategies'. A sufficient condition for admissibility: $\mathbb{E}V_t^2<\infty$ for all $t\in[0,T]$, 
where $V_t$ denotes the portfolio value of the self-financing strategy
\end{itemize}

%\includegraphics[angle=270,width=0.3\textwidth]{gordon-gekko.eps}
%\end{figure}
\end{frame}

\begin{frame}{A first no-arbitrage result: put-call parity}
No arbitrage already pins down some prices \emph{without any model}. Take a European call and put with the same strike $K$ and maturity $T$.
\begin{itemize}
\item at $T$: $\;(S_T-K)^+-(K-S_T)^+=S_T-K$, so \emph{long call $-$ short put} pays exactly $S_T-K$;
\item but \emph{one share $-$ $K$ zero-coupon bonds} pays the same $S_T-K$ at $T$;
\item equal payoffs $\Rightarrow$ equal price today (law of one price):
\end{itemize}
\[ \boxed{\;C_t-P_t = S_t-K\,\e^{-r(T-t)}\;}\qquad\text{(put-call parity).} \]
\textbf{Forward price.} The same reasoning gives the no-arbitrage price for delivery of the stock at $T$:
\[ F_t = S_t\,\e^{r(T-t)} . \]
Neither relation uses the Black-Scholes model - only no arbitrage.
\end{frame}

\begin{frame}{No-arbitrage bounds on the call price}
Even before we price it, no arbitrage bounds a European call:
\[ \big(S_t-K\,\e^{-r(T-t)}\big)^+ \;\le\; C_t \;\le\; S_t . \]
\begin{itemize}
\item \emph{upper bound} $C_t\le S_t$: the call never pays more than the stock, so it cannot cost more (else short the call, buy the stock);
\item \emph{lower bound}: from parity $C_t=P_t+S_t-K\e^{-r(T-t)}\ge S_t-K\e^{-r(T-t)}$ (as $P_t\ge0$), and $C_t\ge0$;
\item since $S_t-K\e^{-r(T-t)}>S_t-K$, the call is worth \emph{more} than its intrinsic value $(S_t-K)^+$ - so on a non-dividend stock an \textbf{American call is never exercised early}.
\end{itemize}
\begin{fftbox}
Try the same for an American \emph{put}: the analogous lower bound $\bigl(K\e^{-r(T-t)}-S_t\bigr)^+$ does \emph{not} dominate the intrinsic value. Can early exercise be optimal? (Hint: $S_t\approx 0$, $r>0$.)
\end{fftbox}
\end{frame}

\section{Toy model}

\begin{frame}{Motivation}
First we discuss a very simple discrete time model and study pricing of options under the assumption that arbitrage opportunities do not exist.
We discuss three methods:
\begin{itemize}
\item pricing by replication
\item risk-neutral pricing
\item the pricing kernel
\end{itemize}
Later on we will discuss their continuous-time analogues.
Although the math gets more complicated, 
the underlying ideas are the same!
\end{frame}


\begin{frame}{Model for financial market}
\begin{itemize}
\item two states-of-the-world: $\Omega=\{ \text{``up''},\text{``down''}\}$
\item two assets: risky asset $S$ and riskless asset $B$
\begin{itemize}
\item $B_1(\text{``up''})=B_1(\text{``down''})=B_1>0$
\item for
\[
S_1(\omega)=\left\{
              \begin{array}{ll}
                s_u, & \hbox{ if } \omega=\text{``up''} \\
                s_d, & \hbox{ if }\omega=\text{``down''}
              \end{array}
            \right.
\]
we assume 
$0<s_d<s_u$
\end{itemize}
\item  probability of state ``up'' is $p\in(0,1)$, i.e. $\mathbb{P}(\text{``up''})=p$
\end{itemize}

\begin{frame}{No arbitrage}
\textbf{Theorem} 
There are no arbitrage opportunities in our financial market if and only if 
\[
0<\frac{s_d}{S_0}<\frac{B_1}{B_0}<\frac{s_u}{S_0}.
\]
\textbf{Proof} \\
`$\Rightarrow$':  exercise \\
 `$\Leftarrow$':  this follows from the risk-neutral construction derived later in this section (verify!)
\\ \vspace{.2cm}
A full proof of `$\Rightarrow$' is given in the \hyperlink{app:noarb}{appendix}.
\end{frame}

\begin{frame}{Method 1: pricing by replication}
\begin{itemize}
\item construct portfolio $V$ with $V_1(\omega)=C_1(\omega)$ for all $\omega$
\begin{itemize}
\item as portfolio generates exactly the same pay-offs as option does (for all states-of-the-world), the portfolio is called \emph{replicating} portfolio
\item Why? As there are no intermediate cashflows, we must have $C_0 = \phi S_0 + \psi B_0$. Otherwise
we would create arbitrage opportunities (this is the \emph{law of one price}). So via the replicating portfolio we find the no-arbitrage price of the option
\end{itemize}
\item notice that we have to determine $\phi$ and $\psi$ such that
\begin{align*}
c_u&= \phi s_u + \psi B_1 \\
c_d&= \phi s_d + \psi B_1
\end{align*}
which yields
\[
\phi=\frac{ c_u -c_d}{s_u-s_d} \text{ and } \psi=\frac{c_d s_u - s_d c_u}{B_1(s_u-s_d)}
\]
\item price of option (at $t=0$):
\[
C_0=\phi S_0 + \psi B_0
\]
\end{itemize}
\end{frame}

\begin{frame}{Method 2: risk-neutral pricing}
\textbf{Motivation}
\begin{itemize}
\item
people would like to have pricing formula of the form:
\[
\frac{A_0}{B_0} = \mathbb{E}\left[ \frac{A_1}{B_1} \right]  
\]
for all assets $A$, i.e. price equals expected discounted cashflows
\item note that
\[
\frac{A_0}{B_0} = \mathbb{E}\left[ \frac{A_1}{B_1} \right] \quad \Longleftrightarrow\quad
 \mathbb{E}\left[ \frac{A_1}{A_0} \right] =\frac{B_1}{B_0}
\]
\item however, already for $A=S$ this almost never holds true in real
financial markets:
\[
 \mathbb{E}\left[ \frac{S_1}{S_0} \right] =\frac{B_1}{B_0},
\]
would mean that the expected (gross) return on the stock equals the riskless return - i.e.\ that investors demand \emph{no risk premium} for bearing risk, which is not what we observe
\end{itemize}
\end{frame}

\begin{frame}{Intermezzo:  a biased coin can save the day} 
\begin{itemize}
\item
Suppose we throw a \emph{fair} coin, $\Omega=\{H,T\}$, and you get
pay-off $X=1$ if $\omega=H$ and $X=0$ if $\omega=T$
\item the price of entering this game is 0.3 (given)
\item (for some reason) we want to have the identity
\[
\text{price} = \mathbb{E}[ X] = \text{Prob}(\{H\})
\]
\item as the coin is fair the equation does not hold true if we use the true, ``real-world'',  probability measure ($0.3\neq 0.5)$
\item however: we can just create \emph{artificial} probability measure
$\mathbb{Q}$ with $\mathbb{Q}(\{H\})=0.3$, which does lead to
\[
\text{price} = \mathbb{E}_{\mathbb{Q}}[X]
\]
\item back in the market, the role of the given ``price $0.3$'' is played by today's asset prices: we will choose $\mathbb{Q}$ so that it reproduces $S_0$ and $B_0$.
\end{itemize}
\end{frame}

\begin{frame}{Method 2: risk-neutral pricing}
\begin{itemize}
\item  people would like to have pricing formula of the form:
\[
\frac{A_0}{B_0} = \mathbb{E}\left[ \frac{A_1}{B_1} \right]\quad   
\Longleftrightarrow\quad
 \mathbb{E}\left[ \frac{A_1}{A_0} \right] =\frac{B_1}{B_0}\quad (\star)
\]
for all assets $A$
\item if we use ``real-world'' probabilities then this identity will
typically not hold for $A=S$
\item could we repair identity by using artificial probability measure $\mathbb{Q}$ instead of ``real-world'' measure $\mathbb{P}$?
\item solving
\[
\frac{S_0}{B_0}=\mathbb{E}_{\mathbb{Q}}
\left[ \frac{S_1}{B_1}\right] = q \frac{s_u}{B_1}  +(1-q) \frac{s_d}{B_1}
\]
yields
\[
q=\frac{\frac{B_1}{B_0}-\frac{s_d}{S_0}    }{ \frac{s_u}{S_0}-\frac{s_d}{S_0}  }
\]
\item if $q\in [0,1]$ then we indeed have found 
artificial probability measure for which $(\star)$ is true by construction
(for $A=B$ the identity is trivial)
\end{itemize}
\end{frame}

\begin{frame}{Method 2: risk-neutral pricing}
Recall earlier theorem: there is no arbitrage if and only if
\[
0<\frac{s_d}{S_0}<\frac{B_1}{B_0}<\frac{s_u}{S_0}
\]
which occurs exactly if $q\in (0,1)$. So we have proved:
\\
\vspace{.3cm}
\textbf{Intermediate result} \\
The market is free of arbitrage opportunities if and only if there exists a probability measure $\mathbb{Q}$ equivalent to $\mathbb{P}$
 such that $S/B$ is a martingale under $\mathbb{Q}$, i.e.
\[ 
 \frac{S_0}{B_0} =\mathbb{E}_{\mathbb{Q}}[ \frac{S_1}{B_1} ].
\]
%\end{itemize}
\end{frame}

\begin{frame}{Method 2: risk-neutral pricing}
Terminology:
\begin{itemize}
\item As $S/B$ is martingale under $\mathbb{Q}$ we have
\[
\frac{S_0}{B_0}=\mathbb{E}_{\mathbb{Q}}\left[ \frac{S_1}{B_1}\right],
\]
which yields
\[
\frac{B_1}{B_0} =\mathbb{E}_{\mathbb{Q}} \left[ \frac{S_1}{S_0} \right] ,
\]
i.e. the expected (gross) return on the risky asset $S$, under $\mathbb{Q}$, is equal to the return/interest on the riskless asset $B$.
\item $\mathbb{Q}$ is often called ``risk-neutral measure''
\begin{itemize}
\item Please note that $\mathbb{Q}$ is \emph{artificial} measure and does not describe actual probabilities!
\end{itemize}
\item $\mathbb{Q}$ is also called \emph{equivalent martingale measure}
\end{itemize}
\end{frame}

\begin{frame}{Method 2: risk-neutral pricing}
So we have repaired the formula  $S_0/B_0 = \mathbb{E}[S_1/B_1]$, but it 
was quite a discussion.  What is the true benefit?
\begin{itemize}
\item introduce, as before, an option to the market (with payoff $C_1$ at $t=1$)
\item we already know that there is a replicating portfolio $(\phi,\psi)$ such that $V_1(\omega) =\phi S_1(\omega) + \psi B_1= C_1(\omega)$ for all $\omega$
\item and, if there are no arbitrage opportunities, $C_0=\phi S_0 + \psi B_0$
\item now note that
\begin{align*}
\mathbb{E}_{\mathbb{Q}}\left[ \frac{C_1} {B_1} \right]
=\phi \mathbb{E}_{\mathbb{Q}}\left[ \frac{S_1} {B_1} \right]
+\psi = \phi \frac{S_0}{B_0} + \psi = \frac{1}{B_0}( \phi S_0 + \psi B_0)=\frac{C_0}{B_0}, 
\end{align*}
so we can calculate for any option (or asset) its price at $t=0$ via the above identity as soon as we know $\mathbb{Q}$! There is no need to calculate 
the replicating portfolios. 
\end{itemize}
\end{frame}

\begin{frame}{Worked example: one-period model}
\textbf{Setup}: $B_0=1$, $B_1=1.02$ (so $r=2\%$), real-world probability $p=\mathbb{P}(\text{up})=0.7$, and the claim is a call with strike $K=50$.
\vspace{0.4cm}
\begin{center}
\begin{tikzpicture}[xscale=3.7,yscale=1.4,>={Stealth[length=2.4mm]}]
\node (s0) at (0,0) {$S_0=125,\ \ B_0=1$};
\node (su) at (1,1) {$S_{1,u}=150,\quad c_u=100$};
\node (sd) at (1,-1) {$S_{1,d}=100,\quad c_d=50$};
\draw[->,thick,gray] (s0)--(su) node[pos=0.5,above,font=\small]{$p=0.7$};
\draw[->,thick,gray] (s0)--(sd) node[pos=0.5,below,font=\small]{$1-p=0.3$};
\end{tikzpicture}
\end{center}
\end{frame}

\begin{frame}{Worked example: pricing by replication}
Solve $c_u=\phi s_u+\psi B_1$ and $c_d=\phi s_d+\psi B_1$:
\begin{align*}
\phi &= \frac{c_u-c_d}{s_u-s_d}=\frac{100-50}{150-100}=1,\\[2pt]
\psi &= \frac{c_d s_u - s_d c_u}{B_1(s_u-s_d)}=\frac{50\cdot150-100\cdot100}{1.02\cdot 50}=-\frac{2500}{51}\approx -49.02 .
\end{align*}
No-arbitrage price of the claim at $t=0$:
\[
C_0=\phi S_0+\psi B_0 = 125 - 49.02 = 75.98 .
\]
\textbf{Check} (the portfolio reproduces the payoff in both states):
\[
\phi s_u+\psi B_1 = 150-50 = 100 = c_u,\qquad
\phi s_d+\psi B_1 = 100-50 = 50 = c_d .
\]
\end{frame}

\begin{frame}{Worked example: risk-neutral pricing}
Risk-neutral probability (makes $S/B$ a $\mathbb{Q}$-martingale):
\[
q=\frac{\frac{B_1}{B_0}-\frac{s_d}{S_0}}{\frac{s_u}{S_0}-\frac{s_d}{S_0}}
=\frac{1.02-0.80}{1.20-0.80}=\frac{0.22}{0.40}=0.55 .
\]
Price as discounted $\mathbb{Q}$-expectation of the payoff:
\[
C_0=\frac{1}{B_1}\mathbb{E}_{\mathbb{Q}}[C_1]=\frac{0.55\cdot100+0.45\cdot50}{1.02}=\frac{77.5}{1.02}=75.98 ,
\]
\emph{identical} to the replication price - as it must be.
\end{frame}

\begin{frame}{Worked example: why a \emph{different} measure?}
The price is \emph{not} the discounted real-world expectation. Under $\mathbb{P}$ ($p=0.7$):
\[
\frac{1}{B_1}\mathbb{E}_{\mathbb{P}}[S_1]=\frac{0.7\cdot150+0.3\cdot100}{1.02}=\frac{135}{1.02}=132.35\neq 125=S_0,
\]
\[
\frac{1}{B_1}\mathbb{E}_{\mathbb{P}}[C_1]=\frac{0.7\cdot100+0.3\cdot50}{1.02}=\frac{85}{1.02}=83.33\neq 75.98 .
\]
Under the risk-neutral $\mathbb{Q}$ ($q=0.55$):
\[
\frac{1}{B_1}\mathbb{E}_{\mathbb{Q}}[S_1]=\frac{0.55\cdot150+0.45\cdot100}{1.02}=\frac{127.5}{1.02}=125=S_0 .
\]
\begin{center}
\emph{Only $\mathbb{Q}$ reproduces today's traded prices; this is exactly why we price under $\mathbb{Q}$ rather than $\mathbb{P}$.}
\end{center}
\end{frame}

\item Price a call with strike $K=100$ (payoffs $c_u=20$, $c_d=0$).
\end{enumerate}
\end{quizbox}
\medskip
\begin{fftbox}
In the worked example the real-world probability $p=0.7$ never entered the price $C_0$ - so two investors who disagree about $p$ still agree on the option price. Where did $p$ go? (Hint: the replication argument never uses it. Compare: the Black-Scholes price does not depend on $\mu$.)
\end{fftbox}
\end{frame}

\begin{frame}{Method 2: risk-neutral pricing}
\textbf{First fundamental theorem of asset pricing (for the toy model)} \\
The market is free of arbitrage opportunities if and only if there exists a probability measure $\mathbb{Q}$ equivalent to $\mathbb{P}$
 such that $A/B$ is a martingale, under $\mathbb{Q}$, for all assets $A$.
\\ \vspace{.5cm}
\textbf{Application} We can price options by applying this theorem twice:
\begin{itemize}
\item Write down model for $B$ and $S$ (without arbitrage opportunities).
\item  Apply theorem (with $A= B$ and $A=S$; note that for $A=B$ the `martingale-property' is  trivial). This yields unique $\mathbb{Q}$.  ($\star$)
\item Now introduce an option to market with payoff $C_1$. We want to set
price such that we do not create arbitrage opportunity.
\item Apply theorem again: $A/B$ should be a martingale for $A=B, S, C$. For $A=S$ we obtain unique $\mathbb{Q}$ from previous step $(\star)$. 
And $A=C$ yields
\[
C_0 = B_0 \mathbb{E}_{\mathbb{Q}}[ C_1/B_1].
\]
\end{itemize}
\end{frame}

\begin{frame}{Method 2: risk-neutral pricing}
\textbf{Proof} \\
`$\Rightarrow$':  exercise - check that we have proved this implication
(via explicit construction of $\mathbb{Q}$) \\
`$\Leftarrow$': \\
Consider portfolio $V$  with $V_1\geq 0$ and $V_1(\omega) >0$ for at least one state-of-the-world. If we prove that $V_0 > 0$, then we have shown that arbitrage opportunities do not exist. \\
Note that we have (why?!)
\begin{align*}
0 &< \mathbb{E}_{\mathbb{Q}}\left[ \frac{V_1}{B_1} \right]
=
\phi\mathbb{E}_{\mathbb{Q}}\left[ \frac{S_1}{B_1} \right] + \psi
\\ 
&= \phi \frac{S_0}{B_0} + \psi = \frac{1}{B_0}(\phi S_0 + \psi B_0)
= \frac{V_0}{B_0},
\end{align*}
which yields $V_0>0$.
\end{frame}

\begin{frame}{Method 3: the pricing kernel}
A third, equivalent route. A \textbf{pricing kernel} (a.k.a.\ stochastic discount factor) is a random variable $m>0$ such that, for \emph{every} asset $A$,
\[
A_0=\mathbb{E}_{\mathbb{P}}[\,m\,A_1\,].
\]
In the toy model $m$ takes two values $m_u,m_d$, determined from the basic assets:
\begin{align*}
B_0&=\mathbb{E}_{\mathbb{P}}[m B_1]=B_1\big(p\,m_u+(1-p)m_d\big),\\
S_0&=\mathbb{E}_{\mathbb{P}}[m S_1]=p\,m_u s_u+(1-p)m_d s_d .
\end{align*}
Two equations, two unknowns. Then \emph{any} claim is priced by
\[
C_0=\mathbb{E}_{\mathbb{P}}[m\,C_1]=p\,m_u c_u+(1-p)m_d c_d .
\]
\end{frame}

\begin{frame}{Method 3: relation to $\mathbb{Q}$, and the worked example}
The three methods are one and the same. Solving the two equations gives
\[
m_\omega=\frac{B_0}{B_1}\,\frac{q_\omega}{p_\omega},
\qquad\text{i.e.}\qquad
m=\underbrace{\frac{B_0}{B_1}}_{\text{discount}}\;\times\;\underbrace{\frac{\rd\mathbb{Q}}{\rd\mathbb{P}}}_{\text{change of measure}} .
\]
For our numbers ($p=0.7$, $q=0.55$, $B_1=1.02$):
\[
m_u=\frac{0.55}{1.02\cdot0.7}\approx0.770,\qquad
m_d=\frac{0.45}{1.02\cdot0.3}\approx1.471,
\]
\[
C_0=0.7\cdot0.770\cdot100+0.3\cdot1.471\cdot50\approx 75.98 ,
\]
again the same price. Note $m_d>m_u$: a payoff in the ``down'' (low-$S$) state is more expensive - the kernel encodes the price of risk.
\end{frame}

\begin{frame}{Multi-period binomial model}

\begin{frame}{Pricing by backward induction}
\textbf{Key observation:} each \emph{one-period sub-tree} is exactly the model we already solved.
\begin{itemize}
\item start at maturity: the values at $t=2$ are the known payoffs;
\item step back one period at a time. At a node at time $t$ with one-period successors ``$u$''/``$d$'', the no-arbitrage value is
\[
V_t=\frac{B_t}{B_{t+1}}\Big(q\,V_{t+1}^{u}+(1-q)\,V_{t+1}^{d}\Big),\qquad
q=\frac{B_{t+1}/B_t-s_d/S_t}{s_u/S_t-s_d/S_t},
\]
equivalently, replicate that single period with $(\phi,\psi)$;
\item the resulting strategy is \emph{self-financing}: at $t=1$ we rebalance, and the value of the old position exactly funds the new one (no cash injected/withdrawn).
\end{itemize}
We illustrate with replication (the spreadsheet also verifies the risk-neutral route).
\end{frame}

\begin{frame}{Two-period example: step back to $t=1$}
At each $t=1$ node we solve a one-period problem for the $t=2$ payoffs.
\\[2pt]
\textbf{Up node} ($S_{1,u}=115\to 130/95$, payoffs $75/40$):
\[
\phi_{1,u}=\frac{75-40}{130-95}=1,\qquad \psi_{1,u}\approx-49.55,\qquad
V_{1,u}\approx 62.97 .
\]
\textbf{Down node} ($S_{1,d}=75\to 95/45$, payoffs $40/0$):
\[
\phi_{1,d}=\frac{40-0}{95-45}=0.8,\qquad \psi_{1,d}\approx-32.43,\qquad
V_{1,d}\approx 25.95 .
\]
\end{frame}

\begin{frame}{Two-period example: step back to $t=0$}
Now treat $V_{1,u}\approx 62.97$ and $V_{1,d}\approx 25.95$ as the ``payoffs'' of a one-period problem at $t=0$:
\[
\phi_0=\frac{62.97-25.95}{115-75}\approx 0.926,\qquad \psi_0\approx-41.41,
\]
\[
V_0=\phi_0 S_0+\psi_0 B_0=0.926\cdot100-41.41\approx 51.16 .
\]
\begin{columns}[c]
\begin{column}{0.46\textwidth}
\footnotesize
So the no-arbitrage price of the two-period call is $\approx 51.16$.\\[3pt]
At $t=1$ we \emph{rebalance} from $(\phi_0,\psi_0)$ to $(\phi_{1,\cdot},\psi_{1,\cdot})$; self-financing means the rebalancing costs nothing.
\end{column}

\begin{frame}{Food for thought}
\begin{fftbox}
Suppose the stock can move to \emph{three} values at $t=1$ (``up'', ``middle'', ``down'') instead of two, with the bond unchanged.
\begin{itemize}
\item Can you still replicate an arbitrary payoff $(c_u,c_m,c_d)$ using only the stock and the bond?
\item How many probability measures $\mathbb{Q}$ make $S/B$ a martingale now?
\item What does this imply for the \emph{uniqueness} of the no-arbitrage price?
\end{itemize}
\end{fftbox}
\vspace{1mm}
{\footnotesize This is the essence of \emph{market incompleteness}: with more states than independent traded assets, replication fails and $\mathbb{Q}$ is no longer unique. We return to completeness in Part~3.}
\end{frame}

\begin{frame}{Toy model - concluding remarks}
For the toy model we have seen that we can price options by
\begin{itemize}
\item constructing a replicating portfolio,
\item using risk-neutral pricing, and
\item using the pricing kernel.
\end{itemize}
All three give the \emph{same} price, and they extend to the multi-period tree via backward induction.
\\
\vspace{.5cm}
For the continuous-time case we will see that exactly the same ideas are used. Unfortunately (?), the math gets a bit more difficult.
\end{frame}

\section{Appendix}

\begin{frame}{Appendix: proof of the no-arbitrage theorem}
\hypertarget{app:noarb}{}
\textbf{Claim.} The market is free of arbitrage $\iff 0<\dfrac{s_d}{S_0}<\dfrac{B_1}{B_0}<\dfrac{s_u}{S_0}$.
\medskip

\textbf{`$\Rightarrow$'} (contrapositive). Suppose the chain of inequalities fails; recall $0<s_d<s_u$.
\begin{itemize}
\item If $\dfrac{B_1}{B_0}\leq \dfrac{s_d}{S_0}$: buy one share, financed by selling $S_0/B_0$ units of the bond, so $V_0=0$. Then
\[
V_1=S_1-\tfrac{S_0}{B_0}B_1\ \geq\ s_d-\tfrac{S_0}{B_0}B_1\ \geq\ 0,
\]
with strict inequality in the ``up'' state $\Rightarrow$ arbitrage.
\item If $\dfrac{B_1}{B_0}\geq \dfrac{s_u}{S_0}$: the reverse strategy (short one share, invest $S_0$ in the bond) gives $V_0=0$ and $V_1=\tfrac{S_0}{B_0}B_1-S_1\geq 0$, strict in the ``down'' state $\Rightarrow$ arbitrage.
\end{itemize}
So absence of arbitrage forces the strict inequalities.
\medskip

\textbf{`$\Leftarrow$'} The inequalities are equivalent to $q\in(0,1)$; the risk-neutral construction (Method~2) then assigns every claim a price consistent with $S$ and $B$, leaving no arbitrage. \hfill$\square$
\end{frame}
