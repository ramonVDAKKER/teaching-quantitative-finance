\section[toc=Black-Scholes PDE]{The Black-Scholes Partial Differential Equation}

\begin{frame}{The problem}
\textbf{Setup:}
\begin{itemize}
\item Black-Scholes market (1 risky asset)
\item we restrict to
Markovian trading strategies:
\[
\phi_t=\tilde\phi(B_t, S_t)= \phi(t,S_t) \text{ and } \psi_t=\tilde\psi(B_t, S_t)
=\psi (t, S_t)
\]
So we can write, for some function $F$,
\[
 V_t=\phi(t, S_t) S_t + \psi(t, S_t) B_t=F(t, S_t)
\]
\end{itemize}
\textbf{Question:} \\
Which $F$'s correspond to \textbf{self-financing} trading strategies?
\\ \vspace{.25cm}
\textbf{Why interesting?}
\begin{itemize}
\item consider option with payoff $h(S_T)$ at maturity
\item if we can find self-financing portfolio with $F(T, S_T)= h(S_T)$ then
no-arbitrage price of option at time $0\leq t <T$ is given by
\[
C_t=F(t, S_t)
\]
\end{itemize}
\end{frame}

\begin{frame}{Black-Scholes Partial Differential Equation}
\textbf{Black-Scholes Partial Differential Equation:} \\
$F$ corresponds to a self-financing trading strategy if and only if $F$ is a solution to the Partial Differential Equation (equation with function $G$ as variable)
\[
\frac{\partial G}{\partial t}(t, s) + r s \frac{\partial G}{\partial
s}(t, s) + \frac{1}{2}\sigma^2 s^2 \frac{\partial^2 G}{\partial
s^2}(t,s) -r G(t,s )=0\,\, \forall s>0, t\in[0, T),
\]
and in that case the positions in the self-financing trading strategy are given by:
\[
\phi(t,s)=\frac{\partial F}{\partial s}(t, s),
\]
and
\[
\psi(t,s)=\frac{F(t,s)-\phi(t,s) s}{\exp(rt)}.
\]
\\ \vspace{.3cm}
\textbf{Remarks:}
\begin{itemize}
\item  $\mu$ does not play a role!
\item using different models (i.e. SDEs)
for $B$ and $S$ leads, in general, to
different PDE!
\end{itemize}
\end{frame}

\begin{frame}{From the binomial tree to the delta}
In the one-period binomial model (part~1) the replicating stock position was a \emph{difference quotient},
\[ \phi=\frac{c_u-c_d}{s_u-s_d}=\frac{F(t+\Delta t,s_u)-F(t+\Delta t,s_d)}{s_u-s_d}, \]
the slope of the value across the two stock outcomes. In the PDE it is $\phi=F_s=\partial F/\partial s$, the \textbf{delta}: as the step shrinks, the secant slope $\to$ the tangent slope.

\begin{frame}{Derivation of the PDE}
Which $F$ are possible (when using self-financing strategies $(\phi(t,S_t),\psi(t,S_t))$)?
We have
\begin{itemize}
\item[(1)] $V_t=F(t,S_t)$
\item[(2)] $\rd V_t= \phi(t,S_t) \rd S_t + \psi(t,S_t) \rd B_t$
\end{itemize}
\vspace{-.2cm}
\begin{itemize}
\item apply It\^{o} to (1) and insert $\rd S_t$  $\implies$
$
\rd V_t=\textcolor{blue}{a_t} \rd t + \textcolor{red}{b_t} \rd W_t
$
\item insert $\rd S_t$ and $\rd B_t$ in RHS of (2) $\implies$
$\rd V_t=\textcolor{blue}{c_t} \rd t + \textcolor{red}{d_t} \rd W_t$
\item obtain system of equations
\[
\left\{
  \begin{array}{ll}
    a_t=c_t  \\
    b_t=d_t % \\
%    (1).
  \end{array}
\right.
\]
\item solving $\implies$ conditions on $F$ (PDE)
\end{itemize}
\end{frame}

\begin{frame}{Warning: sloppy notation}
We will often use the following notations/abbreviations when working with PDEs:
\begin{itemize}
\item $F_s = (\partial F / \partial s) = (\partial F / \partial s)(t,S_t)$
\item $F_t = (\partial F / \partial t) = (\partial F / \partial t)(t,S_t)$
\item $F = F(t,S_t)$
\item etc.
\end{itemize}
The notation $F_t$ is somewhat confusing as it seems to refer to a stochastic process. And you argue that it would be a natural abbreviation for $F(t,S_t)$. \\
If you want to be clear and safe it could be better to just write down the partial derivatives fully.
\end{frame}

\item[(2)] $\rd V_t= \phi(t,S_t) \rd S_t + \psi(t,S_t) \rd B_t$

\end{align*}
\item hence $\phi_t = \phi(t,S_t) = F_{s}(t,S_t)=F_s$
\item and
\begin{align*}
F_s \mu S_t +F_t +  \frac{1}{2}\sigma^2 F_{ss} S_t^2

\end{align*}
\end{itemize}

\begin{frame}{Derivation}
We need to prove
\[
\frac{\partial F}{\partial t}(t,s) + r s \frac{\partial F}{\partial
s}(t,s) + \frac{1}{2}\sigma^2 s^2 \frac{\partial^2 F}{\partial
s^2}(t,s) -r F(t,s)=0\quad \forall s>0,\,t\in[0, T),
\]
On the previous slide we obtained
\begin{align*}
F_t + \frac{1}{2}\sigma^2 F_{ss} S_t^2    
&=  r\psi_t B_t.
\end{align*}
As $\psi_t B_t = F - \phi_t S_t = F - F_s S_t$
we obtain the result.
\end{frame}



\begin{frame}{Alternative Derivation}
\begin{itemize}
\item sell (write) and hold 1 option with price process
$C_t = F(t, S_t)$  (assumption!)
\item let $\phi_t$ be number of stocks at time $t$ and $\psi_t$ the position in the MMA 
\item yields portfolio value $V_t =  - C_t + \phi_t S_t + \psi_t B_t$
\item this portfolio is self-financing if
\[
\rd V_t =  - \rd C_t  + \phi_t \rd S_t + \psi_t \rd B_t
\]
\item 
using It\^o:
\[
\rd V_t = - F_s \rd S_t - F_t \rd t - \frac{1}{2}F_{ss} \rd [S,S]_t + \phi_t \rd S_t
+ \psi_t r B_t \rd t
\]
\item can only eliminate local risk ($\rd W_t$) for $\phi_t = F_s(t, S_t)$!
\item yields $\rd V_t = \cdots \rd t$  which has (locally) no risk, so
rate of return is same as on $B$ $\implies$ $\rd V_t =r V_t \rd t$
\item hence 
\[
-F_t - \frac{1}{2}F_{ss}\sigma^2 S_t^2 + r\psi_t B_t = r \left(  - F  + F_s S_t + \psi_t B_t
\right)
\]
which yields Black-Scholes PDE
\end{itemize}

\end{frame}


\begin{frame}{Application}
Given is a European option with payoff $h(S_T)$ at expiration date/maturity $T$. How can we use the PDE to obtain the price of this option? 
\begin{itemize}
\item Solve the  PDE,
\[
\frac{\partial G}{\partial t}(t, s) + r s \frac{\partial G}{\partial
s}(t, s) + \frac{1}{2}\sigma^2 s^2 \frac{\partial^2 G}{\partial
s^2}(t,s) -r G(t,s )=0,
\]
for all $s>0$, $t\in[0,T)$,
under the boundary condition
\[
G(T,s)=h(s) \text{ for all } s>0.
\]
\item If $F$ is solution to PDE satisfying the boundary condition, then no-arbitrage implies that the price of the option at time $t\in [0, T)$ is given by $F(t, S_t)$.
\item If you are lucky: `closed-form' solution can be found
\begin{itemize}
\item there is relation to well studied  heat equation
\end{itemize}
\item unlucky:
use numerical techniques; see notebook
\end{itemize}
\end{frame}

\begin{frame}{Worked example: verifying a candidate solution}
To check that a function $F$ solves the B-S PDE, just substitute it. Two quick cases (recall $F_t+rsF_s+\tfrac12\sigma^2 s^2 F_{ss}-rF=0$):
\medskip

\textbf{(a) the stock, $F(t,s)=s$:} then $F_t=0,\ F_s=1,\ F_{ss}=0$, so
\[ 0+rs\cdot 1+0-r\,s=0.\quad\checkmark \]
\textbf{(b) a zero-coupon bond paying $K$ at $T$, $F(t,s)=K\e^{-r(T-t)}$:} then $F_s=F_{ss}=0$ and $F_t=rK\e^{-r(T-t)}=rF$, so
\[ rF+0+0-rF=0.\quad\checkmark \]
By \emph{linearity} of the PDE, any combination $a\,s+b\,K\e^{-r(T-t)}$ also solves it - consistent with such a stock+bond portfolio being self-financing.
\\[2pt]
{\small(In the exercises you verify that the digital-call solution $F(t,s)=\e^{-r(T-t)}\Phi(d_2)$ solves the PDE.)}
\end{frame}

\begin{frame}{Solving the PDE numerically: finite differences}
When there is no closed form, solve the PDE on a grid $t_0<\dots<t_N=T$ (step $\Delta t$) and $s_0<\dots<s_M$ (step $\Delta s$). Replace the derivatives by differences:
\[
F_t\approx\frac{F(t+\Delta t,s)-F(t,s)}{\Delta t},\qquad
F_s\approx\frac{F(t,s+\Delta s)-F(t,s-\Delta s)}{2\,\Delta s},
\]
\[
F_{ss}\approx\frac{F(t,s+\Delta s)-2F(t,s)+F(t,s-\Delta s)}{\Delta s^2}.
\]
\begin{itemize}
\item start from the known terminal values $F(T,s)=h(s)$ and step \emph{backward} in time;
\item the explicit scheme writes each $F(t,s_i)$ as a (discounted) weighted average of its three neighbours $F(t+\Delta t,s_{i-1}),\,F(t+\Delta t,s_i),\,F(t+\Delta t,s_{i+1})$ - i.e.\ a \emph{trinomial tree}.
\end{itemize}
An implementation - including the explicit/trinomial scheme and its stability condition - is provided in the accompanying notebook.
\end{frame}

\begin{frame}{Feynman-Kac Theorem}
Let $\alpha(t, x)$, $\beta(t,x)$, and $k(t,x)$ be  functions satisfying `regularity conditions'.
Consider the PDE
\[
\frac{\partial G}{\partial t}(t, x) +  \alpha(t, x) \frac{\partial G}{\partial
x}(t, x) + \frac{1}{2} \beta^2(t,x)  \frac{\partial^2 G}{\partial
x^2}(t,x) - k(t, x)  G(t,x )=0, 
\]
for all $x$ and 
$t\in[0, T)$ with boundary condition $G(T, x) = h(x)$ for all $x$.  

Then the solution is given by:
\[
G(t,x)= \mathbb{E}\left[ 
 \exp\left( -  \int_t^T k(s, X_s) \rd s  \right) h( X_T)
\mid X_t=x
\right],
\]
where $X$ is a stochastic process, on the time interval $[t, T]$, defined via the SDE
\[
\rd X_u = \alpha(u,X_u) \rd u + \beta(u, X_u) \rd W_u
\]
and with starting value $X_t=x$, and
where $W$ is a standard Brownian motion.
\end{frame}



\begin{frame}{Sketch of proof (Feynman-Kac)}
Let $F$ solve the PDE with $F(T,x)=h(x)$, and let $X$ follow $\rd X_u=\alpha(u,X_u)\rd u+\beta(u,X_u)\rd W_u$ on $[t,T]$ with $X_t=x$. Define the discounted process
\[ M_u=\exp\!\Big(-\!\int_t^u k(v,X_v)\,\rd v\Big)\,F(u,X_u),\qquad u\in[t,T]. \]
With $D_u=\exp(-\int_t^u k\,\rd v)$, It\^o's formula gives
\begin{align*}
\rd M_u &= D_u\Big[-k\,F\,\rd u + \big(F_t+\alpha F_x+\tfrac12\beta^2 F_{xx}\big)\rd u + \beta F_x\,\rd W_u\Big]\\
&= D_u\underbrace{\big(F_t+\alpha F_x+\tfrac12\beta^2 F_{xx}-kF\big)}_{=\,0\ \text{(the PDE)}}\rd u \;+\; D_u\,\beta F_x\,\rd W_u .
\end{align*}
So $M$ has \emph{no drift}: it is a (local) martingale, hence - under integrability conditions - a martingale. As $M_t=F(t,x)$ and $M_T=\exp(-\int_t^T k\,\rd v)\,h(X_T)$,
\[ F(t,x)=\mathbb{E}[\,M_T\mid X_t=x\,]=\mathbb{E}\Big[\exp\!\Big(-\!\int_t^T \! k(v,X_v)\,\rd v\Big)h(X_T)\,\Big|\,X_t=x\Big]. \,\square \]
\end{frame}

\begin{frame}{Black-Scholes PDE reconsidered}
The no-arbitrage price, at time $t\in [0,T)$, of a European option with payoff $h(S_T)$ at expiration date $T>0$ is given by
$F(t, S_t)$ where $F$ satisfies
\[
\frac{\partial F}{\partial t}(t,s) + r s \frac{\partial F}{\partial
s}(t,s) + \frac{1}{2}\sigma^2 s^2 \frac{\partial^2 F}{\partial
s^2}(t,s) -r F(t,s)=0\quad \forall s>0,\,t\in[0, T).
\]
Feynman-Kac tells us that we have
\[
F(t, s)= \mathbb{E}\left[ \e^{-r(T-t)}
 h( S_T)
\mid S_t=s
\right],
\]
where $S$ is defined via the SDE
\[
\rd S_u = r S_u \rd u + \sigma S_u \rd W_u, \quad S_t=s,
\]
where $W$ is a standard Brownian motion.
\end{frame}

\begin{frame}{Uniqueness and the maximum principle}
Two structural facts make ``the'' price well-defined and well-behaved.
\begin{itemize}
\item \textbf{Uniqueness.} Under mild growth conditions the B-S PDE together with the terminal condition $F(T,s)=h(s)$ has a \emph{unique} solution. Hence $F(t,S_t)$ is unambiguous - the analytic counterpart of market \emph{completeness}.
\item \textbf{Maximum principle.} For this (parabolic) PDE a solution attains its extreme values on the boundary of the domain - here the terminal time and the $s$-boundaries. Two consequences:
\begin{itemize}
\item \emph{monotonicity}: if $h_1\le h_2$ then $F_1\le F_2$ (prices respect the ordering of payoffs);
\item \emph{no-arbitrage bounds}, e.g.\ $0\le F_{\text{call}}(t,s)\le s$.
\end{itemize}
\end{itemize}
Both are also immediate from Feynman-Kac: $F(t,s)=\mathbb{E}[\e^{-r(T-t)}h(S_T)\mid S_t=s]$ is linear and monotone in the payoff $h$.
\end{frame}
